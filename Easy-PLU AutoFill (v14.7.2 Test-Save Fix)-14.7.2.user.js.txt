    function hohleDaten(key) {
        try {
            const d = localStorage.getItem(key);
            return d ? JSON.parse(d) : null;
        } catch (e) {
            return null;
        }
    }

    function speichereDaten(key, val) {
        try {
            localStorage.setItem(key, JSON.stringify(val));
        } catch (e) {
            console.error('Fehler beim Speichern:', e);
        }
    }

    function sammleAlleTextKnoten(root = document.body) {
        let knoten = [];
        let elemente = Array.from(root.querySelectorAll('*'));

        for (let el of elemente) {
            const tag = el.tagName.toLowerCase();
            if (['script', 'style', 'footer', 'header', 'svg', 'button', 'input', 'nav'].includes(tag)) continue;

            if (tag === 'img') {
                const alt = el.getAttribute('alt');
                const title = el.getAttribute('title');
                const src = el.getAttribute('src');

                if (alt) knoten.push({ el: el, text: alt.trim(), type: 'alt' });
                if (title) knoten.push({ el: el, type: 'title', text: title.trim() });
                if (src && src.includes('/')) {
                    const datei = src.split('/').pop().split('.')[0].replace(/[-_]/g, ' ');
                    knoten.push({ el: el, text: datei.trim(), type: 'src' });
                }
            }

            if (el.children.length === 0) {
                const txt = (el.innerText || el.textContent || '').trim();
                if (txt.length > 0) {
                    knoten.push({ el: el, text: txt, type: 'text' });
                }
            }

            if (el.shadowRoot) {
                knoten = knoten.concat(sammleAlleTextKnoten(el.shadowRoot));
            }
        }
        return knoten;
    }

    function fuehrePluFillAus(isAuto = false) {
        try {
            const gelernteNamenPfade = hohleDaten('mein_name_pfad_index') || [];
            const gelerntePluPfade = hohleDaten('mein_plu_pfad_index') || [];
            const gespeichertePlus = hohleDaten('mein_plu_speicher') || {};
            const alleKnoten = sammleAlleTextKnoten();

            let gefundenerName = null;

            for (let pfad of gelernteNamenPfade) {
                for (let k of alleKnoten) {
                    if (k.text.toLowerCase() === pfad.sampleText?.toLowerCase() ||
                        (k.type === pfad.type && k.el.className && k.el.className === pfad.className)) {
                        gefundenerName = k.text;
                        break;
                    }
                }
                if (gefundenerName) break;
            }

            if (!gefundenerName) {
                for (let k of alleKnoten) {
                    const txtClean = k.text.toLowerCase().trim();
                    if (gespeichertePlus[txtClean]) {
                        gefundenerName = k.text;
                        break;
                    }
                }
            }

            if (!gefundenerName) {
                if (isAuto) return false;
                const manuellerName = prompt('[Lernen] Produktname nicht erkannt. Wie heißt das Produkt?');
                if (!manuellerName) return false;
                gefundenerName = manuellerName.trim();

                let treffer = alleKnoten.find(k => k.text.toLowerCase().includes(gefundenerName.toLowerCase()));
                if (treffer) {
                    gelernteNamenPfade.push({
                        className: treffer.el.className || '',
                        type: treffer.type,
                        sampleText: treffer.text
                    });
                    speichereDaten('mein_name_pfad_index', gelernteNamenPfade);
                }
            }

            const nameClean = gefundenerName.toLowerCase().trim();
            let zielPlu = null;

            if (gespeichertePlus[nameClean]) {
                zielPlu = gespeichertePlus[nameClean];
                zeigeStatus(`Erkannt: ${gefundenerName} -> ${zielPlu}`);
            }
            else {
                for (let pfad of gelerntePluPfade) {
                    for (let k of alleKnoten) {
                        if (k.type === pfad.type && k.el.className === pfad.className) {
                            const m = k.text.match(/\b\d{2,3}\b/);
                            if (m) {
                                zielPlu = m[0];
                                break;
                            }
                        }
                    }
                    if (zielPlu) break;
                }
            }

            if (!zielPlu) {
                if (isAuto) return false;
                const neuePlu = prompt(`Produkt "${gefundenerName}" erkannt.\nBitte PLU-Nummer eingeben:`);
                if (!neuePlu) return false;
                zielPlu = neuePlu.trim();
            }

            gespeichertePlus[nameClean] = zielPlu;
            speichereDaten('mein_plu_speicher', gespeichertePlus);

            let pluTreffer = alleKnoten.find(k => {
                const m = k.text.match(/\b\d{2,3}\b/);
                return m && m[0] === zielPlu;
            });

            if (pluTreffer) {
                const hatBereitsPfad = gelerntePluPfade.some(p => p.className === pluTreffer.el.className);
                if (!hatBereitsPfad) {
                    gelerntePluPfade.push({
                        className: pluTreffer.el.className || '',
                        type: pluTreffer.type
                    });
                    speichereDaten('mein_plu_pfad_index', gelerntePluPfade);
                }
            }

            trageEinUndSende(zielPlu);
            return true;

        } catch (err) {
            console.error('Fehler:', err);
            return false;
        }
    }

    function fuehreTestFillAus(isAuto = false) {
        try {
            const gespeichertePlus = hohleDaten('mein_plu_speicher') || {};
            const alleKnoten = sammleAlleTextKnoten();

            let gefundenerName = null;

            for (let k of alleKnoten) {
                const txtClean = k.text.toLowerCase().trim();
                if (gespeichertePlus[txtClean]) {
                    gefundenerName = k.text;
                    break;
                }
            }

            if (!gefundenerName) {
                if (isAuto) return false;

                // Fallback: Wenn kein bekannter Artikel gefunden wird, fragen wir nach dem Namen und der PLU
                const manuellerName = prompt('[Test-Modus] Artikel nicht im Speicher. Welcher Artikel ist das?');
                if (!manuellerName) return false;
                gefundenerName = manuellerName.trim();
            }

            const nameClean = gefundenerName.toLowerCase().trim();
            let zielPlu = null;

            if (gespeichertePlus[nameClean]) {
                zielPlu = gespeichertePlus[nameClean];
                zeigeStatus(`[Test] Treffer: ${gefundenerName} -> ${zielPlu}`);
            } else {
                if (isAuto) return false;

                // Hier wird jetzt sauber nach der PLU gefragt UND direkt abgespeichert!
                const neuePlu = prompt(`[Test-Modus] Keine PLU für "${gefundenerName}" gefunden.\nBitte PLU-Nummer eingeben (wird gespeichert):`);
                if (!neuePlu) return false;
                zielPlu = neuePlu.trim();

                gespeichertePlus[nameClean] = zielPlu;
                speichereDaten('mein_plu_speicher', gespeichertePlus);
                zeigeStatus(`Neu gespeichert: ${gefundenerName} -> ${zielPlu}`);
            }

            trageEinUndSende(zielPlu);
            return true;
        } catch (err) {
            console.error('Test-Fehler:', err);
            return false;
        }
    }

    function trageEinUndSende(plu) {
        function findeInput(root = document.body) {
            let el = root.querySelector('input');
            if (el) return el;
            for (let child of root.querySelectorAll('*')) {
                if (child.shadowRoot) {
                    let res = findeInput(child.shadowRoot);
                    if (res) return res;
                }
            }
            return null;
        }

        const inputFeld = findeInput();

        if (inputFeld) {
            inputFeld.focus();
            const nativeSetter = Object.getOwnPropertyDescriptor(window.HTMLInputElement.prototype, "value").set;
            nativeSetter.call(inputFeld, plu);
            inputFeld.dispatchEvent(new Event('input', { bubbles: true }));
            inputFeld.dispatchEvent(new Event('change', { bubbles: true }));

            setTimeout(() => {
                ['keydown', 'keypress', 'keyup'].forEach(type => {
                    inputFeld.dispatchEvent(new KeyboardEvent(type, {
                        key: 'Enter', keyCode: 13, code: 'Enter', bubbles: true, cancelable: true
                    }));
                });
            }, 60);
        }
    }
